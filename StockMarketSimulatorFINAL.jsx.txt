                <div>

                    value={authEmail}

                    onKeyPress={(e) => e.key === 'Enter' && handleLogin()}
                    placeholder="user@example.com"

                <div>

                    placeholder="••••••••"

                  />
                </div>
              </div>

              {/* Login Button */}
              <button
                onClick={handleLogin}
                className="w-full bg-gradient-to-r from-cyan-500 to-cyan-600 hover:from-cyan-600 hover:to-cyan-700 text-white font-bold py-3 rounded-lg transition-all transform hover:scale-105"
              >
                {authMode === 'login' ? 'Login' : 'Register & Play'}
              </button>

              {/* Demo Info */}
              <div className="bg-slate-700/30 rounded-lg p-4 text-sm text-slate-300">
                <p className="mb-2">📝 <strong>Demo Account:</strong></p>
                <p className="text-xs">Email: demo@apex.com</p>

  // MAIN APP
  return (

      {/* Animated background */}

                </p>
              </div>

                <p className="text-lg font-bold">{currentUser?.email}</p>
                <button
                  onClick={handleLogout}
                  className="text-red-400 hover:text-red-300 text-sm mt-1 flex items-center gap-1"
                >
                  <LogOut size={14} /> Logout

              </div>
            </div>
          </div>

          {/* Navigation Tabs */}
          <div className="flex gap-4 border-b border-slate-700/50 overflow-x-auto">
            {['market', 'portfolio', 'watchlist', 'screener', 'orders', 'alerts'].map(tab => (

        {/* Market Tab */}

            {/* Main Chart Area */}

                </div>

              </div>

              <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
                {[
                  { label: 'P/E Ratio', value: selectedStockData?.pe },
                  { label: 'EPS', value: '$' + selectedStockData?.eps },
                  { label: 'Dividend', value: selectedStockData?.dividend ? '$' + selectedStockData?.dividend : 'N/A' },
                  { label: 'Avg Volume', value: (selectedStockData?.avgVolume / 1000000).toFixed(1) + 'M' },
                ].map((stat, idx) => (
                  <div key={idx} className="bg-slate-800/50 backdrop-blur-xl border border-slate-700/50 rounded-xl p-4">
                    <p className="text-slate-400 text-sm mb-1">{stat.label}</p>
                    <p className="text-xl font-bold text-cyan-400">{stat.value}</p>

                  <p className="text-lg font-bold">{selectedStockData?.sector}</p>

              <div className="bg-slate-800/50 backdrop-blur-xl border border-slate-700/50 rounded-xl p-4">
                <div className="relative mb-4">
                  <Search className="absolute left-3 top-3 text-slate-400" size={18} />

                    value={searchQuery}

                    className="w-full bg-slate-700/50 border border-slate-600 rounded-lg pl-10 pr-4 py-2 text-white placeholder-slate-500 focus:outline-none focus:border-cyan-500"
                  />
                </div>

                {/* Filters */}
                <div className="space-y-3">
                  <div>
                    <p className="text-xs text-slate-400 mb-2">Sector</p>
                    <select
                      value={sector}
                      onChange={(e) => setSector(e.target.value)}
                      className="w-full bg-slate-700/50 border border-slate-600 rounded px-2 py-1 text-sm text-white focus:outline-none focus:border-cyan-500"
                    >
                      {sectors.map(s => (
                        <option key={s} value={s}>{s === 'all' ? 'Semua Sector' : s}</option>
                      ))}
                    </select>
                  </div>
                  <div>
                    <p className="text-xs text-slate-400 mb-2">Range Harga: ${priceRange.min} - ${priceRange.max}</p>
                    <input
                      type="range"
                      min="0"
                      max="5000"
                      step="100"
                      value={priceRange.max}

                      className="w-full"
                    />
                  </div>
                </div>
              </div>

              {/* Stock List */}
              <div className="bg-slate-800/50 backdrop-blur-xl border border-slate-700/50 rounded-xl overflow-hidden">
                <div className="p-4 border-b border-slate-700/50">
                  <h3 className="font-bold text-cyan-400 mb-3">MARKET MOVERS</h3>
                  <div className="flex gap-2 flex-wrap">
                    {['change', 'volume'].map(sort => (
                      <button
                        key={sort}
                        onClick={() => setSortBy(sort)}
                        className={`text-xs px-2 py-1 rounded transition-all ${
                          sortBy === sort
                            ? 'bg-cyan-500/30 text-cyan-400'
                            : 'bg-slate-700/30 text-slate-400 hover:bg-slate-700/50'
                        }`}
                      >
                        {sort === 'change' ? '% Chg' : 'Vol'}
                      </button>
                    ))}
                  </div>
                </div>

                <div className="max-h-[600px] overflow-y-auto">
                  {filteredStocks.map(([symbol, stock]) => (
                    <div
                      key={symbol}
                      onClick={() => setSelectedStock(symbol)}
                      className={`p-4 border-b border-slate-700/30 cursor-pointer transition-all ${
                        selectedStock === symbol
                          ? 'bg-cyan-500/20 border-l-4 border-l-cyan-500'
                          : 'hover:bg-slate-700/30'

                          <p className="font-bold">{symbol}</p>
                          <p className="text-xs text-slate-400 truncate">{stock.name}</p>

                          onClick={(e) => {

                            toggleWatchlist(symbol);

                          className="text-slate-400 hover:text-cyan-400"
                        >
                          {watchlist.includes(symbol) ? <Eye size={16} /> : <EyeOff size={16} />}
                        </button>
                      </div>

                        <span className="text-base font-bold text-cyan-400">${stock.price}</span>

                        </span>

              </button>

                    {Object.entries(portfolio).map(([symbol, quantity]) => {

                      const boughtValue = orderHistory
                        .filter(o => o.symbol === symbol && o.type === 'BUY')
                        .reduce((sum, o) => sum + o.total, 0);
                      const gain = totalValue - boughtValue;

                      return (

                            <div>
                              <p className="font-bold text-lg">{symbol}</p>

                                    price: stock.price,

                          <span className="text-2xl font-bold text-cyan-400">${stock.price}</span>
                          <span className={`font-semibold text-sm ${stock.change >= 0 ? 'text-green-400' : 'text-red-400'}`}>
                            {stock.change > 0 ? '+' : ''}{stock.change}%
                          </span>
                        </div>
                      </div>
                    </div>
                  );
                })}
              </div>
            )}
          </div>
        )}

        {/* Screener Tab */}
        {activeTab === 'screener' && (
          <div className="bg-slate-800/50 backdrop-blur-xl border border-slate-700/50 rounded-xl p-6">
            <h2 className="text-2xl font-bold mb-6">Stock Screener</h2>
            <div className="overflow-x-auto">
              <table className="w-full text-sm">
                <thead className="border-b border-slate-700">
                  <tr>
                    <th className="text-left py-3 px-4 text-slate-400 font-semibold">Symbol</th>
                    <th className="text-left py-3 px-4 text-slate-400 font-semibold">Name</th>
                    <th className="text-right py-3 px-4 text-slate-400 font-semibold">Price</th>
                    <th className="text-right py-3 px-4 text-slate-400 font-semibold">Change %</th>
                    <th className="text-right py-3 px-4 text-slate-400 font-semibold">P/E</th>
                    <th className="text-right py-3 px-4 text-slate-400 font-semibold">Dividend</th>
                    <th className="text-left py-3 px-4 text-slate-400 font-semibold">Sector</th>
                  </tr>
                </thead>
                <tbody>
                  {filteredStocks.slice(0, 20).map(([symbol, stock]) => (
                    <tr key={symbol} className="border-b border-slate-700/30 hover:bg-slate-700/20 cursor-pointer"
                      onClick={() => {
                        setSelectedStock(symbol);
                        setActiveTab('market');
                      }}
                    >
                      <td className="py-3 px-4 font-bold text-cyan-400">{symbol}</td>
                      <td className="py-3 px-4 text-slate-300">{stock.name}</td>
                      <td className="text-right py-3 px-4 font-bold">${stock.price}</td>
                      <td className={`text-right py-3 px-4 font-bold ${stock.change >= 0 ? 'text-green-400' : 'text-red-400'}`}>
                        {stock.change > 0 ? '+' : ''}{stock.change}%
                      </td>
                      <td className="text-right py-3 px-4">{stock.pe}</td>
                      <td className="text-right py-3 px-4">{stock.dividend || 'N/A'}</td>
                      <td className="py-3 px-4">{stock.sector}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {/* Orders Tab */}
        {activeTab === 'orders' && (
          <div className="bg-slate-800/50 backdrop-blur-xl border border-slate-700/50 rounded-xl p-6">
            <h2 className="text-2xl font-bold mb-6">Order History</h2>
            {orderHistory.length === 0 ? (
              <div className="text-slate-400 text-center py-12">
                <Clock className="mx-auto mb-4 opacity-50" size={48} />
                <p>Tidak ada orders.</p>
              </div>
            ) : (
              <div className="overflow-x-auto">
                <table className="w-full text-sm">
                  <thead className="border-b border-slate-700">
                    <tr>
                      <th className="text-left py-3 px-4 text-slate-400 font-semibold">Symbol</th>
                      <th className="text-left py-3 px-4 text-slate-400 font-semibold">Type</th>
                      <th className="text-right py-3 px-4 text-slate-400 font-semibold">Qty</th>
                      <th className="text-right py-3 px-4 text-slate-400 font-semibold">Price</th>
                      <th className="text-right py-3 px-4 text-slate-400 font-semibold">Total</th>
                      <th className="text-left py-3 px-4 text-slate-400 font-semibold">Time</th>
                    </tr>
                  </thead>
                  <tbody>
                    {orderHistory.map(order => (
                      <tr key={order.id} className="border-b border-slate-700/30 hover:bg-slate-700/20">
                        <td className="py-3 px-4 font-bold">{order.symbol}</td>
                        <td className="py-3 px-4">
                          <span className={`px-3 py-1 rounded text-xs font-semibold ${
                            order.type === 'BUY'
                              ? 'bg-green-500/20 text-green-400'
                              : 'bg-red-500/20 text-red-400'
                          }`}>
                            {order.type}
                          </span>
                        </td>
                        <td className="text-right py-3 px-4">{order.quantity}</td>
                        <td className="text-right py-3 px-4 text-cyan-400">${order.price}</td>
                        <td className="text-right py-3 px-4 font-semibold">${order.total.toFixed(2)}</td>
                        <td className="py-3 px-4 text-slate-400 text-xs">{order.timestamp}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
            )}
          </div>
        )}

        {/* Alerts Tab */}
        {activeTab === 'alerts' && (
          <div className="space-y-6">
            <div className="flex gap-4">
              <button
                onClick={() => setShowAlertModal(true)}
                className="bg-gradient-to-r from-cyan-500 to-cyan-600 hover:from-cyan-600 hover:to-cyan-700 text-white font-bold py-2 px-6 rounded-lg transition-all"
              >
                + Alert Baru
              </button>
            </div>

            <div className="bg-slate-800/50 backdrop-blur-xl border border-slate-700/50 rounded-xl p-6">
              <h2 className="text-2xl font-bold mb-6">Price Alerts</h2>
              {alerts.length === 0 ? (
                <div className="text-slate-400 text-center py-12">
                  <Bell className="mx-auto mb-4 opacity-50" size={48} />
                  <p>Tidak ada alerts.</p>
                </div>
              ) : (
                <div className="space-y-4">
                  {alerts.map(alert => (
                    <div key={alert.id} className="bg-slate-700/30 rounded-lg p-4 flex items-center justify-between">
                      <div>
                        <p className="font-bold text-lg">{alert.symbol}</p>
                        <p className="text-sm text-slate-400">
                          Alert ketika harga {alert.type === 'above' ? 'naik diatas' : 'turun dibawah'} ${alert.price}
                        </p>
                      </div>
                      <button
                        onClick={() => setAlerts(alerts.filter(a => a.id !== alert.id))}
                        className="text-red-400 hover:text-red-300 font-semibold"
                      >
                        Hapus
                      </button>
                    </div>
                  ))}
                </div>
              )}
            </div>
          </div>
        )}
      </main>

      {/* Buy Modal */}
      {showBuyModal && (
        <div className="fixed inset-0 bg-black/50 backdrop-blur-sm flex items-center justify-center z-50 p-4">
          <div className="bg-slate-900 border border-slate-700 rounded-xl max-w-md w-full p-8">
            <h3 className="text-2xl font-bold mb-6">Beli {selectedStock}</h3>

            <div className="space-y-6">
              <div>
                <p className="text-slate-400 text-sm mb-2">Market Price</p>
                <p className="text-4xl font-bold text-cyan-400">${selectedStockData?.price}</p>
              </div>

              <div>
                <label className="block text-slate-400 text-sm mb-3">Order Type</label>
                <div className="grid grid-cols-3 gap-2">
                  {['market', 'limit'].map(type => (
                    <button
                      key={type}
                      onClick={() => setOrderType(type)}
                      className={`px-3 py-2 rounded font-semibold text-sm transition-all ${
                        orderType === type
                          ? 'bg-cyan-500 text-white'
                          : 'bg-slate-700/50 text-slate-300'
                      }`}
                    >
                      {type.charAt(0).toUpperCase() + type.slice(1)}
                    </button>
                  ))}
                </div>
              </div>

              {orderType === 'limit' && (
                <div>
                  <label className="block text-slate-400 text-sm mb-2">Limit Price</label>
                  <input
                    type="number"
                    value={limitPrice}
                    onChange={(e) => setLimitPrice(e.target.value)}
                    className="w-full bg-slate-800 border border-slate-600 rounded px-4 py-2 text-white focus:outline-none focus:border-cyan-500"
                  />
                </div>
              )}

              <div>
                <label className="block text-slate-400 text-sm mb-3">Quantity</label>
                <div className="flex items-center gap-4">
                  <button
                    onClick={() => setBuyQuantity(Math.max(1, buyQuantity - 1))}
                    className="bg-slate-700/50 hover:bg-slate-700 p-2 rounded transition-all"
                  >
                    <Minus size={20} />
                  </button>
                  <input
                    type="number"
                    value={buyQuantity}
                    onChange={(e) => setBuyQuantity(Math.max(1, parseInt(e.target.value) || 1))}
                    className="flex-1 bg-slate-800 border border-slate-600 rounded px-4 py-2 text-center font-bold text-white"
                  />
                  <button
                    onClick={() => setBuyQuantity(buyQuantity + 1)}
                    className="bg-slate-700/50 hover:bg-slate-700 p-2 rounded transition-all"
                  >
                    <Plus size={20} />
                  </button>
                </div>
              </div>

              <div className="bg-slate-800/50 rounded-lg p-4 space-y-3">
                <div className="flex justify-between">
                  <span className="text-slate-400">Total Cost</span>
                  <span className="font-bold text-cyan-400 text-lg">
                    ${((orderType === 'limit' ? parseFloat(limitPrice) || selectedStockData?.price : selectedStockData?.price) * buyQuantity).toFixed(2)}
                  </span>
                </div>
              </div>

              <div className="flex gap-3">
                <button
                  onClick={() => {
                    setShowBuyModal(false);
                    setOrderType('market');
                    setLimitPrice('');
                  }}
                  className="flex-1 bg-slate-700 hover:bg-slate-600 px-4 py-3 rounded-lg font-semibold"
                >
                  Cancel
                </button>
                <button
                  onClick={handleBuyStock}
                  disabled={balance < (orderType === 'limit' ? parseFloat(limitPrice) || selectedStockData?.price : selectedStockData?.price) * buyQuantity}
                  className="flex-1 bg-gradient-to-r from-cyan-500 to-cyan-600 hover:from-cyan-600 hover:to-cyan-700 disabled:from-slate-600 disabled:to-slate-600 text-white px-4 py-3 rounded-lg font-semibold"
                >
                  Place Order
                </button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* Alert Modal */}
      {showAlertModal && (
        <div className="fixed inset-0 bg-black/50 backdrop-blur-sm flex items-center justify-center z-50 p-4">
          <div className="bg-slate-900 border border-slate-700 rounded-xl max-w-md w-full p-8">
            <h3 className="text-2xl font-bold mb-6">Set Price Alert</h3>

            <div className="space-y-6">
              <div>
                <label className="block text-slate-400 text-sm mb-2">Stock</label>
                <p className="text-xl font-bold text-cyan-400">{selectedStock}</p>
              </div>

              <div>
                <label className="block text-slate-400 text-sm mb-3">Alert Type</label>
                <div className="flex gap-2">
                  {['above', 'below'].map(type => (
                    <button
                      key={type}
                      onClick={() => setAlertType(type)}
                      className={`flex-1 px-3 py-2 rounded font-semibold text-sm transition-all ${
                        alertType === type
                          ? 'bg-cyan-500 text-white'
                          : 'bg-slate-700/50 text-slate-300'
                      }`}
                    >
                      {type === 'above' ? '↑ Naik' : '↓ Turun'}
                    </button>
                  ))}
                </div>
              </div>

              <div>
                <label className="block text-slate-400 text-sm mb-2">Alert Price</label>
                <input
                  type="number"
                  step="0.01"
                  value={alertPrice}
                  onChange={(e) => setAlertPrice(e.target.value)}
                  className="w-full bg-slate-800 border border-slate-600 rounded px-4 py-2 text-white focus:outline-none focus:border-cyan-500"
                  placeholder="Enter price"
                />
              </div>

              <div className="flex gap-3">
                <button
                  onClick={() => {
                    setShowAlertModal(false);
                    setAlertPrice('');
                  }}
                  className="flex-1 bg-slate-700 hover:bg-slate-600 px-4 py-3 rounded-lg font-semibold"
                >
                  Cancel
                </button>
                <button
                  onClick={handleAddAlert}
                  className="flex-1 bg-gradient-to-r from-cyan-500 to-cyan-600 hover:from-cyan-600 hover:to-cyan-700 text-white px-4 py-3 rounded-lg font-semibold"
                >
                  Set Alert
                </button>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
